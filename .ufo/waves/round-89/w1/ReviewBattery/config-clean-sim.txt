    def limine_clean(data: bytes) -> bytes:
        buf = bytearray(data); size = len(buf); i = 0
        while i < size:
            skip = 0
            trailing = False
            if buf[i] in (0x20, 0x09):
                while i + skip < size and buf[i + skip] in (0x20, 0x09):
                    skip += 1
                if i + skip < size and buf[i + skip] == 0x0A:
                    trailing = True          # goto skip_loop
                else:
                    skip = 0
            if not trailing:
                while i + skip < size and (buf[i + skip] == 0x0D or
                      ((i == 0 or buf[i-1] == 0x0A) and buf[i+skip] in (0x20, 0x09))):
                    skip += 1
            if skip:
                for j in range(i, size - skip):
                    buf[j] = buf[j + skip]
                size -= skip; del buf[size:]
                i -= 1
            i += 1
        return bytes(buf[:size])
