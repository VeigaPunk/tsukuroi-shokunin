def fam_refusal_versions_9digit(c):
    """r95 C1 (QUALITY 139, the ACCEPTED leg): the largest in-bound member of the gate's published
    1-9-digit component cap — `999999999.9.9`, first component 999,999,999 — is a real version: the
    gate must accept it (rc 1 for this fixture's mismatch pin, never rc 64) and select the era from
    it (999999999 >= 12.9.5 -> era B, so the first-line-entry key is invisible -> `panic`). An
    over-strict refusal bound (`<= 8`) refuses the value and fails; a gate that accepts it but reads
    era A answers `prompt` and fails. The r94 roster has no case whose version is an in-bound
    9-digit value, so that bend read `met` 98/98 before this family existed."""
    _era_first_line_panic(c, REFUSAL_DIGIT9)

def _era_first_line_panic(c, version, pass_version=True):
    """V2: the file's FIRST physical line is an entry line and its body carries a mismatch pin and,
    after the body, a `hash_mismatch_panic: no`. Pre-12.9.5 the cut is skipped (no bytes precede
    the '/') so the key IS visible -> `prompt`; from 12.9.5 on the label-end NUL leaves the global
    section empty so the key is INVISIBLE -> `panic`."""
    c.version = version
    c.pass_version = pass_version
    c.add("/%s" % rname(c.rng, ""))
    c.add("  protocol: efi")
    c.pin("LP-MISMATCH", indent="  ")
    set_panic(c, c.rng.choice(["no", "No", "NO"]), indent="")
    efi_entry(c)
    c.pin("OK")
