# STRICT C1 SL-row counter (v2) — defined by seat M1Mutation (round 67), stricter
# than the frozen `grep -cE '^\| SL[0-9]+ \|'`.
# A line counts as ONE valid row iff ALL of:
#   S1 not inside a fenced code block (first non-space chars "```" toggle the fence)
#   S2 exact frozen anchor: line starts with "| SL<digits> |" (byte-equivalent to ^\| SL[0-9]+ \|)
#   S3 exactly 3 visible columns (awk -F'|': NF == 5, i.e. leading+trailing empty fields);
#      a markdown-escaped "\|" inside a cell is protected before the split, so a legitimate
#      row whose evidence quotes a pipeline still yields 3 columns (v1 lacked this)
#   S4 id = SL<digits>, unique across all valid rows
#   S5 claim cell non-empty after trim
#   S6 evidence cell non-empty after trim and matching a line-number pattern:
#        PATH:[0-9]+(-[0-9]+)?            e.g. scrub.c:1640  or scrub.c:1640-1649
#        [0-9][0-9][0-9]-[0-9][0-9][0-9]  e.g. 597-604
#        [0-9][0-9][0-9][0-9]-[0-9][0-9][0-9][0-9]  e.g. 1640-1649
# Output: "STRICT_COUNT=<n>" on stdout (n = distinct valid rows); exit 0.
# -v verbose=1 : per-candidate ACCEPT/REJECT with reason on stderr.

function trim(s) { sub(/^[ \t]+/, "", s); sub(/[ \t]+$/, "", s); return s }

function unprotect(s) { gsub("\001", "|", s); return s }

BEGIN { fence = 0; count = 0 }

{
    raw = $0

    # S1 fence tracking
    tmp = raw
    sub(/^[ \t]+/, "", tmp)
    if (substr(tmp, 1, 3) == "```") { fence = !fence; next }
    if (fence) { if (verbose) printf "REJECT line %d: inside fenced code block\n", NR; next }

    # S2 frozen anchor, reproduced without regex metacharacter hazards
    if (substr(raw, 1, 2) != "| ") next
    rest = substr(raw, 3)
    if (rest !~ /^SL[0-9]/) next
    id = rest
    sub(/[^0-9A-Za-z_].*$/, "", id)
    if (id !~ /^SL[0-9]+$/) next
    if (substr(rest, length(id) + 1, 1) != " ") next
    if (substr(rest, length(id) + 2, 1) != "|") next

    # protect markdown-escaped pipes so S3 sees the real column count
    line = raw
    gsub(/\\\|/, "\001", line)

    # S3 exactly 3 visible columns
    nf = split(line, c, "|")
    if (nf != 5) {
        if (verbose) printf "REJECT line %d (%s): NF=%d, want 5 (3 visible columns)\n", NR, id, nf
        next
    }
    claim = trim(unprotect(c[3]))
    ev = trim(unprotect(c[4]))

    # S4 unique id
    if (id in seen) {
        if (verbose) printf "REJECT line %d (%s): duplicate id\n", NR, id
        next
    }
    # S5 claim non-empty
    if (claim == "") {
        if (verbose) printf "REJECT line %d (%s): empty claim cell\n", NR, id
        next
    }
    # S6 evidence present and line-number bearing
    if (ev == "") {
        if (verbose) printf "REJECT line %d (%s): empty evidence cell\n", NR, id
        next
    }
    hit = (ev ~ /[[:alnum:]_./-]+:[0-9][0-9]*(-[0-9][0-9]*)?/) \
          || (ev ~ /[0-9][0-9][0-9]-[0-9][0-9][0-9]/) \
          || (ev ~ /[0-9][0-9][0-9][0-9]-[0-9][0-9][0-9][0-9]/)
    if (!hit) {
        if (verbose) printf "REJECT line %d (%s): evidence has no line-number pattern: %s\n", NR, id, ev
        next
    }

    seen[id] = 1
    count++
    if (verbose) printf "ACCEPT line %d (%s)\n", NR, id
}

END { printf "STRICT_COUNT=%d\n", count }
--- end strict.awk ---
