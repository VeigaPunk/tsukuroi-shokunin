88: def show_text(lane, rev, path, override):
89:     if override:
90:         try:
91:             with open(override, 'rb') as fh:
92:                 return fh.read().decode('utf-8', errors='replace')

99: def section(lines, head):
100:     start = next((i for i, ln in enumerate(lines) if ln.strip() == head), None)
101:     if start is None:
102:         return None
103:     end = next((k for k in range(start + 1, len(lines)) if lines[k].startswith('## ')), len(lines))
104:     return lines[start:end]

107: def cells(line):
108:     parts = line.split('|')
109:     return [p.strip() for p in parts[1:-1]]

112: def content_legs(adj, anchors, registry, siblings_root):
113:     """Resolve every OUT id the ADJ rows name against the sibling bytes: pin digest, heading,
114:     quote, and the quote at the sibling's live tip. Returns counts + per-id detail."""
115:     ids = sorted({o for cs in adj.values() for o in OUT_REF.findall(cs[-1] if cs else '')})
116:     pin_ok = 0
117:     live_ok = 0
118:     detail = {}
119:     for out in ids:
120:         d = {'pin': False, 'live': False, 'why': []}
121:         anc = anchors.get(out)
122:         rr = registry.get(out)
123:         if anc is None or rr is None:
124:             d['why'].append('no anchor row or registry row')
125:             detail[out] = d
126:             continue
127:         sib = rr.get('sibling') or (anc[2].split('/PROVINCE.md')[0].strip('`').replace('-shokunin', ''))
128:         repo = os.path.join(siblings_root, f'{sib}-shokunin')
129:         commit = str(rr.get('sibling_commit', ''))
130:         want_sha = str(rr.get('sibling_blob_sha256', ''))
131:         quote = anc[4].strip('`').strip() if len(anc) > 4 else ''
132:         heading = anc[3].strip('`').strip() if len(anc) > 3 else ''
133:         if not os.path.isdir(os.path.join(repo, '.git')):
134:             d['why'].append(f'sibling repo absent: {repo}')
135:             detail[out] = d
136:             continue
137:         blob = subprocess.run(['git', '-C', repo, 'show', f'{commit}:PROVINCE.md'],
138:                               capture_output=True, stdin=subprocess.DEVNULL)
139:         if blob.returncode != 0:
140:             d['why'].append(f'pin {commit} does not resolve to PROVINCE.md')
141:             detail[out] = d
142:             continue
143:         text = blob.stdout.decode('utf-8', errors='replace')
144:         if hashlib.sha256(blob.stdout).hexdigest() != want_sha:
145:             d['why'].append('registry blob sha256 does not match the pinned blob')
146:         elif heading not in text:
147:             d['why'].append('heading absent from the pinned blob')
148:         elif quote not in text:
149:             d['why'].append('quote absent from the pinned blob')
150:         else:
151:             d['pin'] = True
152:         tip = subprocess.run(['git', '-C', repo, 'rev-parse', 'HEAD'],
153:                              capture_output=True, stdin=subprocess.DEVNULL)
154:         if tip.returncode == 0:
155:             tipblob = subprocess.run(['git', '-C', repo, 'show',
156:                                       f'{tip.stdout.decode().strip()}:PROVINCE.md'],
157:                                      capture_output=True, stdin=subprocess.DEVNULL)
158:             if tipblob.returncode == 0 and quote and quote in tipblob.stdout.decode('utf-8', errors='replace'):
159:                 d['live'] = True
160:             else:
161:                 d['why'].append('quote absent from the sibling tip bytes')
162:         else:
163:             d['why'].append('sibling HEAD unreadable')
164:         pin_ok += int(d['pin'])
165:         live_ok += int(d['live'])
166:         detail[out] = d
167:     return pin_ok, live_ok, detail

194: def shape_legs(adj, anchors, registry):
195:     """The round-49 legs, factored so the stub can be scored by them without touching the lane."""
196:     bare = linked = anchors_ok = registry_ok = 0
197:     for row_id in sorted(adj):
198:         cs = adj[row_id]
199:         own = cs[-1] if cs else ''
200:         bare += len(BARE.findall(own))
201:         outs = OUT_ID.findall(own)
202:         if outs:
203:             linked += 1
204:         a_ok = r_ok = bool(outs)
205:         for out in outs:
206:             anc = anchors.get(out)
207:             if anc is None or len(anc) < 5:
208:                 a_ok = r_ok = False
209:                 continue
210:             if anc[1] != row_id:
211:                 a_ok = False
212:             if not SIB.fullmatch(anc[2]) or not HEAD.fullmatch(anc[3]):
213:                 a_ok = False
214:             quote = anc[4].strip('`').strip()
215:             if not quote or len(quote.split()) > 25:
216:                 a_ok = False
217:             rr = registry.get(out)
218:             if rr is None:
219:                 r_ok = False
220:                 continue
221:             if not HEX64.fullmatch(str(rr.get('sibling_blob_sha256', ''))):
222:                 r_ok = False
223:             if not HEX12.fullmatch(str(rr.get('sibling_commit', ''))):
224:                 r_ok = False
225:         anchors_ok += int(a_ok)
226:         registry_ok += int(r_ok)
227:     return {'adj_rows': len(adj), 'linked_rows': linked, 'bare_ranges': bare,
228:             'anchors_ok': anchors_ok, 'registry_ok': registry_ok}

279:         own = cs[-1] if cs else ''

287:             row_problems.append('no OUT anchor named')

291:         r_ok = bool(outs)

302:             if anc[1] != row_id:

305:             sm = SIB.fullmatch(anc[2])
