# receipt C — echo guard: hostile unit names planted in a fixture directory
$ mkdir -p $S/fixtures/etc/systemd/system; ln -s /dev/null ...
$ ls -1b $S/fixtures/etc/systemd/system
a=b.service
evil.service\nMASKED\ count=999
evil|pipe.service
legit.service
tab\tstop.service

$ TSUKUROI_UNIT_DIRS=$S/fixtures/etc/systemd/system bash probe.sh --reading masked
BLOCKED reading=MASKED reason=bad-unit-name
rc=1

# the reading emits exactly one line and the forged payload never appears as a line
$ ... | wc -l                                   -> 1
$ ... | grep -c '^MASKED count='                -> 0
$ ... | grep -c 'count=999'                     -> 0
$ ... | grep -c 'reason=bad-unit-name'          -> 1
