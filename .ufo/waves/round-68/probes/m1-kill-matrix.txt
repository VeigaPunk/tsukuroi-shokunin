#!/usr/bin/env python3
import sys
sys.dont_write_bytecode = True
import re

def count(path):
    with open(path, "r", encoding="utf-8", errors="replace") as fh:
        lines = fh.read().splitlines()
    fence = False
    ids = set()
    for raw in lines:
        stripped = raw.lstrip()
        if stripped.startswith("```"):
            fence = not fence
            continue
        if fence:
            continue
        if not raw.startswith("|"):
            continue
        cells = [c.strip() for c in raw.split("|")]
        if len(cells) < 4 or not cells[3]:
            continue
        nonempty = [c for c in cells if c]
        if not nonempty or not nonempty[-1]:
            continue
        if not re.fullmatch(r"RW[0-9]+", cells[1]):
            continue
        ids.add(cells[1])
    return len(ids)

def main(argv):
    if len(argv) < 2:
        sys.stderr.write("usage: mycount.py FILE...\n")
        return 2
    for path in argv[1:]:
        print("%s RW_ROWS=%d" % (path, count(path)))
    return 0

if __name__ == "__main__":
    sys.exit(main(sys.argv))
