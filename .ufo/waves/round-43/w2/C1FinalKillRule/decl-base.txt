### analyze_tree
analyze_tree () 
{ 
    local tree=$1 sys new version late overlay ply kind tgt;
    [ -f "$tree/init" ] || return 1;
    [ -f "$tree/VERSION" ] || return 1;
    sys=$(grep -oF '/sysroot' "$tree/init" | wc -l | tr -d '[:space:]');
    new=$(grep -oF '/new_root' "$tree/init" | wc -l | tr -d '[:space:]');
    version=$(head -n 1 "$tree/VERSION" | tr -d '[:space:]');
    late='-';
    if [ -f "$tree/config" ]; then
        late=$(sed -n 's/^LATEHOOKS="\(.*\)"$/\1/p' "$tree/config" | head -n 1);
    fi;
    [ -n "$late" ] || late='-';
    overlay='-';
    if [ -f "$tree/hooks/btrfs-overlayfs" ]; then
        overlay=$(sed -n 's/^[[:space:]]*root_mnt="\([^"]*\)".*/\1/p' "$tree/hooks/btrfs-overlayfs" | head -n 1);
    fi;
    [ -n "$overlay" ] || overlay='-';
    ply='-';
    if [ -f "$tree/hooks/plymouth" ]; then
        ply=$(sed -n 's/.*--new-root-dir=\([^[:space:]"]*\).*/\1/p' "$tree/hooks/plymouth" | head -n 1);
    fi;
    [ -n "$ply" ] || ply='-';
    if [ -L "$tree/new_root" ]; then
        tgt=$(readlink -- "$tree/new_root");
        case $tgt in 
            sysroot | ./sysroot | /sysroot)
                kind='symlink->sysroot'
            ;;
            *)
                kind="symlink->$tgt"
            ;;
        esac;
    else
        if [ -d "$tree/new_root" ]; then
            kind='directory';
        else
            if [ -e "$tree/new_root" ]; then
                kind='other';
            else
                kind='absent';
            fi;
        fi;
    fi;
    printf '%s\t%s\t%s\t%s\t%s\t%s\t%s\n' "$sys" "$new" "$version" "$late" "$overlay" "$ply" "$kind";
    return 0
}
### audit
audit () 
{ 
    local conf=$1;
    local -a labels=() efis=() shas=() b2oks=() unames=() versions=() sysrefs=() newrefs=();
    local -a kinds=() lates=() overlays=() plies=() rflags=() snaps=() snapros=();
    local -a sels=() mtimes=() kernels=() states=();
    local n=0 l e b c ch ok;
    local rootflags sn ident s computed u tup a bb cc dd ee ff kk sel mtime;
    local btime newer;
    local i booted hassnap sel_flag verdict summary rc;
    local booted_n=0 aliased_n=0 pre42_n=0 untested_n=0 notkernel_n=0 unknown_n=0 mismatch_n=0;
    while IFS='' read -r l e b c ch ok; do
        ENTRY_IDX=$n;
        IMG_COPY='';
        labels[n]=$l;
        efis[n]=$e;
        rootflags=$(cmdline_token "$c" rootflags);
        rflags[n]=$rootflags;
        sn=$(snapshot_number "$rootflags");
        snaps[n]=${sn:--};
        shas[n]='-';
        b2oks[n]='-';
        unames[n]='-';
        versions[n]='-';
        sysrefs[n]=0;
        newrefs[n]=0;
        kinds[n]='-';
        lates[n]='-';
        overlays[n]='-';
        plies[n]='-';
        snapros[n]='-';
        sels[n]=false;
        mtimes[n]='-';
        kernels[n]=0;
        states[n]=ok;
        if [ "$ok" != 1 ]; then
            states[n]=error;
        else
            if ! stage_image "$e"; then
                states[n]=error;
            else
                ident=$(image_identity "$e");
                case $ident in 
                    error)
                        states[n]=error
                    ;;
                    *)
                        :
                    ;;
                esac;
                if [ "${states[n]}" = ok ]; then
                    if s=$(entry_sha256 "$e"); then
                        shas[n]=$s;
                    else
                        states[n]=error;
                    fi;
                fi;
                if [ "${states[n]}" = ok ]; then
                    if [ -n "$b" ]; then
                        if computed=$(entry_b2sum "$e") && [ "${#computed}" -eq 128 ] && [ "$computed" = "${b,,}" ]; then
                            b2oks[n]=ok;
                        else
                            b2oks[n]=MISMATCH;
                        fi;
                    fi;
                    if [ "$ident" = kernel ]; then
                        kernels[n]=1;
                        if u=$(entry_uname "$e"); then
                            unames[n]=$u;
                        fi;
                        if extract_entry "$e"; then
                            if tup=$(analyze_tree "$EXTRACT_DIR"); then
                                IFS='	' read -r a bb cc dd ee ff kk <<< "$tup";
                                sysrefs[n]=$a;
                                newrefs[n]=$bb;
                                versions[n]=$cc;
                                lates[n]=$dd;
                                overlays[n]=$ee;
                                plies[n]=$ff;
                                kinds[n]=$kk;
                            else
                                states[n]=error;
                            fi;
                            guarded_rm "$EXTRACT_DIR";
                            EXTRACT_DIR='';
                        else
                            states[n]=error;
                        fi;
                        if [ "${states[n]}" = ok ]; then
                            mtimes[n]=$(mtime_for "$e");
                            sels[n]=$(entry_selected "$ch");
                            if [ -n "$sn" ]; then
                                snapros[n]=$(snapshot_ro "$e" "$(snapshot_host_path "$rootflags")");
                            fi;
                        fi;
                    fi;
                fi;
                if [ -n "$IMG_COPY" ]; then
                    guarded_rm "$IMG_COPY";
                    IMG_COPY='';
                fi;
            fi;
        fi;
        n=$((n + 1));
    done < <(parse_limine_conf "$conf");
    newer=$(boot_inputs_newer);
    btime=$(btime_epoch);
    printf 'label\tefi\tsha256\tb2_ok\tuname\tVERSION\tinit_sysroot_refs\tinit_new_root_refs\timage_new_root\tlatehooks\toverlay_root\tplymouth_root\trootflags\tsnapshot\tsnapshot_ro\tselected\tverdict\n';
    for ((i = 0; i < n; i++))
    do
        booted=0;
        if [ "${states[i]}" = error ]; then
            verdict=EVALUATION-ERROR;
            unknown_n=$((unknown_n + 1));
        else
            hassnap=0;
            [ "${snaps[i]}" != '-' ] && hassnap=1;
            if [ "${kernels[i]}" = 1 ] && [ "${sels[i]}" = true ] && [ "$newer" = 0 ] && [[ ${mtimes[i]} =~ ^[0-9]+$ ]] && [[ $btime =~ ^[0-9]+$ ]] && [ "${mtimes[i]}" -lt "$btime" ]; then
                booted=1;
            fi;
            if [ "${sels[i]}" = true ]; then
                sel_flag=1;
            else
                sel_flag=0;
            fi;
            verdict=$(verdict_for "${kernels[i]}" "$booted" "$sel_flag" "$hassnap" "${sysrefs[i]}" "${newrefs[i]}" "${kinds[i]}" "${overlays[i]}" "${plies[i]}");
            case $verdict in 
                BOOTED-OBSERVED)
                    booted_n=$((booted_n + 1))
                ;;
                PATH-ALIASED-EMULATED)
                    aliased_n=$((aliased_n + 1))
                ;;
                CONSISTENT-PRE42)
                    pre42_n=$((pre42_n + 1))
                ;;
                UNTESTED)
                    untested_n=$((untested_n + 1))
                ;;
                NOT-A-KERNEL-ENTRY)
                    notkernel_n=$((notkernel_n + 1))
                ;;
            esac;
            if [ "$verdict" = UNTESTED ] && { 
                [ "$hassnap" = 1 ] || [ "$sel_flag" = 1 ]
            }; then
                unknown_n=$((unknown_n + 1));
            fi;
        fi;
        [ "${b2oks[i]}" = MISMATCH ] && mismatch_n=$((mismatch_n + 1));
        printf '%s\t%s\t%s\t%s\t%s\t%s\t%s\t%s\t%s\t%s\t%s\t%s\t%s\t%s\t%s\t%s\t%s\n' "${labels[i]}" "${efis[i]}" "${shas[i]}" "${b2oks[i]}" "${unames[i]}" "${versions[i]}" "${sysrefs[i]}" "${newrefs[i]}" "${kinds[i]}" "${lates[i]}" "${overlays[i]}" "${plies[i]}" "${rflags[i]}" "${snaps[i]}" "${snapros[i]}" "${sels[i]}" "$verdict";
    done;
    local kernel_eval=$((n - notkernel_n - unknown_n));
    if [ "$unknown_n" -gt 0 ] || [ "$n" -eq 0 ] || [ "$kernel_eval" -le 0 ]; then
        summary=UNKNOWN;
        rc=2;
    else
        summary=CONSISTENT;
        rc=0;
    fi;
    printf '# counts entries=%d kernels=%d booted=%d path-aliased=%d pre42=%d untested=%d not-kernel=%d unknown=%d b2_mismatch=%d\n' "$n" "$kernel_eval" "$booted_n" "$aliased_n" "$pre42_n" "$untested_n" "$notkernel_n" "$unknown_n" "$mismatch_n";
    printf 'TSUKUROI_BOOTCHAIN: %s\n' "$summary";
    return "$rc"
}
### boot_inputs_newer
boot_inputs_newer () 
{ 
    if fixture_mode; then
        local f;
        f="$BOOTCHAIN_FIXTURE/boot_inputs_newer";
        if [ -f "$f" ]; then
            tr -d '[:space:]' < "$f";
        else
            printf '0\n';
        fi;
        return 0;
    fi;
    local btime;
    btime=$(btime_epoch);
    if [ -z "$btime" ]; then
        printf '1\n';
        return 0;
    fi;
    if find /etc/fstab /etc/crypttab /etc/mkinitcpio.conf /etc/mkinitcpio.conf.d /etc/default/limine /etc/limine-snapper-sync.conf /usr/lib/initcpio /usr/lib/modules -newermt "@$btime" -print -quit 2> /dev/null | grep -q .; then
        printf '1\n';
    else
        printf '0\n';
    fi
}
### btime_epoch
btime_epoch () 
{ 
    if fixture_mode; then
        local f;
        f="$BOOTCHAIN_FIXTURE/btime";
        if [ -f "$f" ]; then
            tr -d '[:space:]' < "$f";
        else
            printf -- '-\n';
        fi;
        return 0;
    fi;
    awk '/^btime /{print $2; exit}' /proc/stat 2> /dev/null
}
### cleanup
cleanup () 
{ 
    [ -n "$WORK" ] && guarded_rm "$WORK"
}
### cmdline_token
cmdline_token () 
{ 
    local cmdline=$1 key=$2 tok;
    local -;
    set -f;
    for tok in $cmdline;
    do
        case $tok in 
            "$key"=*)
                printf '%s\n' "${tok#*=}";
                return 0
            ;;
        esac;
    done;
    printf '\n'
}
### entry_b2sum
entry_b2sum () 
{ 
    if fixture_mode; then
        local f;
        f=$(fx_dir "$1")/b2sum;
        [ -f "$f" ] || return 1;
        tr -d '\n' < "$f";
        return 0;
    fi;
    b2sum -- "$IMG_COPY" 2> /dev/null | awk '{print $1}'
}
### entry_selected
entry_selected () 
{ 
    local chain=$1 cand_dot sel;
    cand_dot=${chain//\//.};
    while IFS= read -r sel; do
        [ -n "$sel" ] || continue;
        if [ "$sel" = "$chain" ] || [ "$sel" = "$cand_dot" ]; then
            printf 'true\n';
            return 0;
        fi;
    done < <(loader_entries);
    printf 'false\n'
}
### entry_sha256
entry_sha256 () 
{ 
    if fixture_mode; then
        local f;
        f=$(fx_dir "$1")/sha256;
        [ -f "$f" ] || return 1;
        tr -d '\n' < "$f";
        return 0;
    fi;
    sha256sum -- "$IMG_COPY" 2> /dev/null | awk '{print $1}'
}
### entry_uname
entry_uname () 
{ 
    if fixture_mode; then
        tr -d '\n' < "$(fx_dir "$1")/uname";
        return 0;
    fi;
    tr -d '\0\n' < "$WORK/img/$ENTRY_IDX.uname"
}
### extract_entry
extract_entry () 
{ 
    local efi=$1 d img;
    EXTRACT_DIR='';
    d=$(mktemp -d "$WORK/tree.XXXXXX") || return 1;
    if fixture_mode; then
        if cp -R "$(fx_dir "$efi")/extract/." "$d/" 2> /dev/null; then
            EXTRACT_DIR=$d;
            return 0;
        fi;
        guarded_rm "$d";
        return 1;
    fi;
    img=$(work_path "img/$ENTRY_IDX.initrd") || { 
        guarded_rm "$d";
        return 1
    };
    if ! objcopy -O binary --only-section=.initrd "$IMG_COPY" "$img" 2> /dev/null; then
        guarded_rm "$d";
        return 1;
    fi;
    if ( cd -- "$d" && TMPDIR="$WORK" lsinitcpio -x "$img" > /dev/null 2>&1 ); then
        EXTRACT_DIR=$d;
        return 0;
    fi;
    guarded_rm "$d";
    return 1
}
### fixture_mode
fixture_mode () 
{ 
    [ -n "$BOOTCHAIN_FIXTURE" ]
}
### fx_dir
fx_dir () 
{ 
    printf '%s/%s\n' "$BOOTCHAIN_FIXTURE" "$(basename -- "$1")"
}
### guarded_rm
guarded_rm () 
{ 
    local p=$1;
    [ -n "$p" ] || return 0;
    case $p in 
        "${WORK:-/nonexistent-work}" | "${WORK:-/nonexistent-work}"/* | "${SELFTEST_TMP:-/nonexistent-selftest}" | "${SELFTEST_TMP:-/nonexistent-selftest}"/*)
            [ -e "$p" ] || return 0;
            chmod -R u+rwX -- "$p" 2> /dev/null || true;
            rm -rf -- "$p"
        ;;
        *)
            printf '%s: refusing rm of non-work path: %s\n' "$PROG" "$p" 1>&2;
            return 1
        ;;
    esac
}
### host_efi

}
### image_identity
image_identity () 
{ 
    if fixture_mode; then
        local d;
        d=$(fx_dir "$1");
        if [ -f "$d/unreadable" ]; then
            printf 'error\n';
        else
            if [ -f "$d/not-a-uki" ]; then
                printf 'not-kernel\n';
            else
                if [ -s "$d/uname" ]; then
                    printf 'kernel\n';
                else
                    if [ -f "$d/uname" ]; then
                        printf 'not-kernel\n';
                    else
                        printf 'error\n';
                    fi;
                fi;
            fi;
        fi;
        return 0;
    fi;
    local dst;
    [ -n "$IMG_COPY" ] || { 
        printf 'error\n';
        return 0
    };
    dst=$(work_path "img/$ENTRY_IDX.uname") || { 
        printf 'error\n';
        return 0
    };
    if objcopy -O binary --only-section=.uname "$IMG_COPY" "$dst" 2> /dev/null && [ -s "$dst" ]; then
        printf 'kernel\n';
    else
        case "$1" in 
            *[Ee][Ff][Ii]/[Ll]inux/*)
                printf 'error\n'
            ;;
            *)
                printf 'not-kernel\n'
            ;;
        esac;
    fi
}
### init_work
init_work () 
{ 
    local created;
    created=$(mktemp -d "${TMPDIR:-/tmp}/tsukuroi-bootchain.XXXXXX") || return 1;
    chmod 700 -- "$created" 2> /dev/null || { 
        rm -rf -- "$created";
        return 1
    };
    case $created in 
        /*)

        ;;
        *)
            rm -rf -- "$created";
            return 1
        ;;
    esac;
    WORK=$created;
    mkdir -p -- "$WORK/img" "$WORK/copy" || { 
        guarded_rm "$WORK";
        return 1
    };
    trap cleanup EXIT;
    trap 'exit 2' INT TERM HUP;
    return 0
}
### loader_entries
loader_entries () 
{ 
    local v found=0;
    if fixture_mode; then
        [ -f "$BOOTCHAIN_FIXTURE/loader_entries" ] && cat -- "$BOOTCHAIN_FIXTURE/loader_entries";
        return 0;
    fi;
    for v in /sys/firmware/efi/efivars/LoaderEntrySelected-* /sys/firmware/efi/efivars/LimineLastBootedEntry-*;
    do
        [ -f "$v" ] || continue;
        found=1;
        tail -c +5 -- "$v" 2> /dev/null | tr -d '\0';
        printf '\n';
    done;
    if [ "$found" -eq 0 ] && [ -d /sys/firmware/efi/efivars ]; then
        printf '%s: no LoaderEntrySelected/LimineLastBootedEntry efivar readable; every entry reports selected=false\n' "$PROG" 1>&2;
    fi
}
### main
main () 
{ 
    local mode='';
    while [ $# -gt 0 ]; do
        case $1 in 
            --live)
                mode=live
            ;;
            --selftest)
                mode=selftest
            ;;
            --help | -h)
                usage;
                return 0
            ;;
            *)
                printf '%s: unknown argument: %s\n' "$PROG" "$1" 1>&2;
                usage 1>&2;
                return 2
            ;;
        esac;
        shift;
    done;
    case $mode in 
        selftest)
            run_selftest;
            return $?
        ;;
        live)
            if [ -n "$BOOTCHAIN_FIXTURE" ]; then
                printf '%s: refusing --live while BOOTCHAIN_FIXTURE is set\n' "$PROG" 1>&2;
                printf 'TSUKUROI_BOOTCHAIN: UNKNOWN\n';
                return 2;
            fi;
            if [ "${EUID:-1}" -eq 0 ]; then
                printf '%s: refusing --live as root: the audit must run unprivileged\n' "$PROG" 1>&2;
                printf 'TSUKUROI_BOOTCHAIN: UNKNOWN\n';
                return 2;
            fi
        ;;
        *)
            printf '%s: one of --live, --selftest, --help is required\n' "$PROG" 1>&2;
            usage 1>&2;
            printf 'TSUKUROI_BOOTCHAIN: UNKNOWN\n';
            return 2
        ;;
    esac;
    local missing='' c;
    for c in sudo cat objcopy lsinitcpio sha256sum b2sum find awk sed grep basename mktemp stat tail tr head wc readlink uname chmod rm;
    do
        command -v "$c" > /dev/null 2>&1 || missing="$missing $c";
    done;
    if [ -n "$missing" ]; then
        printf '%s: missing required command(s):%s\n' "$PROG" "$missing" 1>&2;
        printf 'TSUKUROI_BOOTCHAIN: UNKNOWN\n';
        return 2;
    fi;
    if ! init_work; then
        printf '%s: cannot create a private mktemp work dir (TMPDIR=%s); refusing\n' "$PROG" "${TMPDIR:-/tmp}" 1>&2;
        printf 'TSUKUROI_BOOTCHAIN: UNKNOWN\n';
        return 2;
    fi;
    local conf=$WORK/limine.conf;
    if ! sudo -n cat -- "$LIVE_CONF" > "$conf" 2> /dev/null; then
        printf '%s: cannot read %s through sudo -n; refusing to guess\n' "$PROG" "$LIVE_CONF" 1>&2;
        printf 'TSUKUROI_BOOTCHAIN: UNKNOWN\n';
        return 2;
    fi;
    audit "$conf";
    return $?
}
### mk_init
mk_init () 
{ 
    local tree=$1 sys=$2 new=$3 i;
    mkdir -p -- "$tree";
    : > "$tree/init";
    for ((i = 0; i < sys; i++))
    do
        printf '# mount /sysroot %d\n' "$i" >> "$tree/init";
    done;
    for ((i = 0; i < new; i++))
    do
        printf '# mount /new_root %d\n' "$i" >> "$tree/init";
    done
}
### mk_uki_fx
mk_uki_fx () 
{ 
    local d=$1 sha=$2 b2=$3 uname=$4 version=$5 sys=$6 new=$7 overlay=$8 ply=$9;
    local kind=${10:-} snapro=${11:-} mtime=${12:-};
    mkdir -p -- "$d/extract/hooks";
    printf '%s\n' "$sha" > "$d/sha256";
    printf '%s\n' "$b2" > "$d/b2sum";
    printf '%s\n' "$uname" > "$d/uname";
    printf '%s\n' "$version" > "$d/extract/VERSION";
    mk_init "$d/extract" "$sys" "$new";
    printf 'LATEHOOKS="plymouth btrfs-overlayfs"\n' > "$d/extract/config";
    printf '#!/usr/bin/ash\nrun_latehook() {\n\troot_mnt="%s"\n}\n' "$overlay" > "$d/extract/hooks/btrfs-overlayfs";
    printf '#!/usr/bin/ash\nrun_latehook(){\n  plymouth update-root-fs --new-root-dir=%s\n}\n' "$ply" > "$d/extract/hooks/plymouth";
    case $kind in 
        symlink)
            mkdir -p -- "$d/extract/sysroot";
            ln -s sysroot "$d/extract/new_root"
        ;;
        dir)
            mkdir -p -- "$d/extract/new_root"
        ;;
        other)
            printf 'x\n' > "$d/extract/new_root"
        ;;
        'symlink->'*)
            ln -s "${kind#symlink->}" "$d/extract/new_root"
        ;;
        absent)
            :
        ;;
        *)
            :
        ;;
    esac;
    [ -n "$snapro" ] && printf '%s\n' "$snapro" > "$d/snapshot_ro";
    [ -n "$mtime" ] && printf '%s\n' "$mtime" > "$d/mtime";
    return 0
}
### mtime_for
mtime_for () 
{ 
    if fixture_mode; then
        local f;
        f=$(fx_dir "$1")/mtime;
        if [ -f "$f" ]; then
            tr -d '[:space:]' < "$f";
        else
            printf -- '-\n';
        fi;
        return 0;
    fi;
    if [[ ${STAGED_MTIME:-} =~ ^[0-9]+$ ]]; then
        printf '%s\n' "$STAGED_MTIME";
    else
        printf -- '-\n';
    fi
}
### parse_limine_conf
parse_limine_conf () 
{ 
    local conf=$1 raw line slashes depth name label chain efi b2 cmdline rest pathok;
    local cur_label='' cur_efi='' cur_b2='' cur_cmdline='' cur_chain='' cur_pathok='';
    local pending_cmdline='' have=0 i;
    local -a stack_label=() stack_name=();
    while IFS= read -r raw || [ -n "$raw" ]; do
        line=${raw#"${raw%%[![:space:]]*}"};
        line=${line%"${line##*[![:space:]]}"};
        case $line in 
            '' | '#'*)
                continue
            ;;
        esac;
        case $line in 
            path:*)
                if [ "$have" = 1 ]; then
                    printf '%s\001%s\001%s\001%s\001%s\001%s\n' "$cur_label" "$cur_efi" "$cur_b2" "$cur_cmdline" "$cur_chain" "$cur_pathok";
                fi;
                rest=${line#path:};
                rest=${rest#"${rest%%[![:space:]]*}"};
                if [[ $rest =~ ^boot\(\):/([A-Za-z0-9._/-]+)(#[0-9a-f]{128})?$ ]] && [[ ${BASH_REMATCH[1]} != *..* ]]; then
                    efi="/${BASH_REMATCH[1]}";
                    b2=${BASH_REMATCH[2]#\#};
                    pathok=1;
                else
                    efi=$rest;
                    b2='';
                    pathok=0;
                fi;
                cur_label=$label;
                cur_efi=$efi;
                cur_b2=$b2;
                cur_cmdline=$pending_cmdline;
                cur_chain=$chain;
                cur_pathok=$pathok;
                pending_cmdline='';
                have=1
            ;;
            cmdline:*)
                cmdline=${line#cmdline:};
                cmdline=${cmdline#"${cmdline%%[![:space:]]*}"};
                if [ "$have" = 1 ]; then
                    cur_cmdline=$cmdline;
                else
                    pending_cmdline=$cmdline;
                fi
            ;;
            /*)
                if [ "$have" = 1 ]; then
                    printf '%s\001%s\001%s\001%s\001%s\001%s\n' "$cur_label" "$cur_efi" "$cur_b2" "$cur_cmdline" "$cur_chain" "$cur_pathok";
                    have=0;
                    pending_cmdline='';
                fi;
                slashes=${line%%[!/]*};
                depth=${#slashes};
                name=${line#"$slashes"};
                name=${name#+};
                [ -n "$name" ] || continue;
                stack_label[depth]=$line;
                stack_name[depth]=$name;
                if [ "$depth" -ge 3 ]; then
                    label="${stack_label[depth - 1]:-} $line";
                else
                    label=$line;
                fi;
                chain='';
                for ((i = 1; i <= depth; i++))
                do
                    [ -n "${stack_name[i]:-}" ] && chain+="${chain:+/}${stack_name[i]}";
                done
            ;;
        esac;
    done < "$conf";
    if [ "$have" = 1 ]; then

### snapshot_host_path
snapshot_host_path () 
{ 
    local rf=$1 sub;
    sub=${rf#subvol=};
    sub=${sub%%,*};
    sub=${sub#/};
    sub=${sub#@};
    sub=${sub#/};
    printf '/%s\n' "$sub"
}
### snapshot_number
snapshot_number () 
{ 
    local rf=$1;
    if [[ $rf =~ \.snapshots/([0-9]+)/snapshot ]]; then
        printf '%s\n' "${BASH_REMATCH[1]}";
    fi;
    return 0
}
### snapshot_ro
snapshot_ro () 
{ 
    if fixture_mode; then
        local f;
        f=$(fx_dir "$1")/snapshot_ro;
        if [ -f "$f" ]; then
            tr -d '[:space:]' < "$f";
        else
            printf -- '-\n';
        fi;
        return 0;
    fi;
    local out;
    command -v btrfs > /dev/null 2>&1 || { 
        printf -- '-\n';
        return 0
    };
    if ! out=$(btrfs property get -t subvol "$2" ro 2> /dev/null); then
        printf -- '-\n';
        return 0;
    fi;
    case $out in 
        *ro=true*)
            printf 'true\n'
        ;;
        *ro=false*)
            printf 'false\n'
        ;;
        *)
            printf -- '-\n'
        ;;
    esac
}
### st_check
st_check () 
{ 
    st_total=$((st_total + 1));
    if [ "$2" = "$3" ]; then
        st_pass=$((st_pass + 1));
    else
        st_fail=$((st_fail + 1));
        printf 'FAIL %s: expected [%s] got [%s]\n' "$1" "$2" "$3";
    fi
}
### st_mk_fallback_fx
st_mk_fallback_fx () 
{ 
    local d=$1;
    mkdir -p -- "$d";
    printf 'deadbeef\n' > "$d/sha256";
    printf 'deadbeef\n' > "$d/b2sum";
    : > "$d/not-a-uki"
}
### stage_image
stage_image () 
{ 
    if fixture_mode; then
        IMG_COPY='';
        [ -f "$(fx_dir "$1")/unreadable" ] && return 1;
        return 0;
    fi;
    local c m1 m2;
    c=$(work_path "copy/$ENTRY_IDX.efi") || return 1;
    STAGED_MTIME='';
    m1=$(sudo -n stat -c %Y -- "$(host_efi "$1")" 2> /dev/null) || m1='';
    if sudo -n cat -- "$(host_efi "$1")" > "$c" 2> /dev/null; then
        m2=$(sudo -n stat -c %Y -- "$(host_efi "$1")" 2> /dev/null) || m2='';
        if [ -n "$m1" ] && [ "$m1" != "$m2" ]; then
            return 1;
        fi;
        STAGED_MTIME=$m1;
        IMG_COPY=$c;
        return 0;
    fi;
    return 1
}
### usage
usage () 
{ 
    awk 'NR == 1 {next} /^#/ {sub(/^# ?/, ""); print; next} {exit}' "$0"
}
### verdict_for
verdict_for () 
{ 
    local kernel=$1 booted=$2 selected=$3 hassnap=$4 sys=$5 new=$6 kind=$7 overlay=$8 ply=$9;
    if [ "$kernel" != 1 ]; then
        printf 'NOT-A-KERNEL-ENTRY\n';
        return 0;
    fi;
    if [ "$booted" = 1 ]; then
        printf 'BOOTED-OBSERVED\n';
        return 0;
    fi;
    if [ "$hassnap" != 1 ] && [ "$selected" != 1 ]; then
        printf 'UNTESTED\n';
        return 0;
    fi;
    if [ "$sys" -gt 0 ] && [ "$new" -eq 0 ] && [ "$kind" = 'symlink->sysroot' ] && [ "$overlay" = /new_root ] && [ "$ply" = /new_root ]; then
        printf 'PATH-ALIASED-EMULATED\n';
        return 0;
    fi;
    if [ "$new" -gt 0 ] && [ "$sys" -eq 0 ] && [ "$kind" = directory ] && [ "$overlay" = /new_root ] && [ "$ply" = /new_root ]; then
        printf 'CONSISTENT-PRE42\n';
        return 0;
    fi;
    printf 'UNTESTED\n'
}
### work_path
work_path () 
{ 
    local rel=$1;
    case $rel in 
        '' | /* | *..*)
            printf '%s: refusing unsafe work path: %s\n' "$PROG" "$rel" 1>&2;
            return 1
        ;;
    esac;
    printf '%s/%s\n' "$WORK" "$rel"
}
